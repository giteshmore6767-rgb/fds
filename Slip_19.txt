Q2 OPTION 2
import numpy as np
a=np.array([1,2,3,4,5]); b=np.array([2,4,6,8,10])
c=np.correlate(a-a.mean(),b-b.mean(),mode='full')
print('Correlation',c.max()/(np.std(a)*np.std(b)*len(a)))
print('Lag',np.argmax(c)-(len(a)-1))
