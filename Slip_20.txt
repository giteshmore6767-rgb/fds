Q2 OPTION 1
import numpy as np
from scipy.stats import zscore
a=np.array([10,12,11,13,12,14,11,13,100],float)
q1,q3=np.percentile(a,[25,75]); i=q3-q1
iq=a[(a<q1-1.5*i)|(a>q3+1.5*i)]
z=a[np.abs(zscore(a))>3]
med=np.median(a); mz=0.6745*(a-med)/np.median(np.abs(a-med))
m=a[np.abs(mz)>3.5]
print(iq,z,m)
b=a[~np.isin(a,iq)]
print('Before',a.mean(),np.median(a),a.std())
print('After',b.mean(),np.median(b),b.std())
