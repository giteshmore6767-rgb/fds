import pandas as pd
from sklearn.preprocessing import LabelEncoder
df=pd.read_csv('StudentData.csv')
for c in ['Age','Fees']: df[c]=df[c].fillna(df[c].mean())
df=pd.get_dummies(df,columns=['City'])
df['Admitted']=LabelEncoder().fit_transform(df['Admitted'])
print(df)
