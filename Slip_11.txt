Q2 OPTION 1
import pandas as pd
import matplotlib.pyplot as plt
df=pd.read_csv('iris.csv')
print(df['species'].value_counts().plot(kind='bar')); plt.show()
for c in df.select_dtypes('number'):
    for name,g in df.groupby('species'):
        plt.hist(g[c],alpha=.5,label=name)
    plt.title(c); plt.legend(); plt.show()

import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt
df=pd.read_csv('iris.csv')
for c in df.select_dtypes('number'):
    sns.boxplot(data=df,x='species',y=c); plt.show()
