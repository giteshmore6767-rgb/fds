import numpy as np
a=np.array([[2,4,6],[3,5,7]],float); w=np.array([1,2,3])
print('Weighted arithmetic',np.average(a,weights=w,axis=1))
print('Weighted harmonic',w.sum()/np.sum(w/a,axis=1))
print('Weighted geometric',np.exp(np.average(np.log(a),weights=w,axis=1)))
